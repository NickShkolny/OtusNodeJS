# CLI-приложение для развертывания Telegram бота с использованием Node.js, MongoDB и Nginx (Диплой домашнего задания №6 с 10 урока)

## Введение

Мы создаем CLI-приложение на Node.js, которое автоматизирует процесс развертывания Telegram бота, включая установку необходимых зависимостей, настройку Nginx, получение SSL-сертификатов от Let's Encrypt с помощью Certbot и запуск приложения. Для хранения переменных окружения будем использовать файл конфигурации.

## 1. Создание файла конфигурации

Сначала создадим файл `config.json`, который будет содержать все необходимые переменные окружения:

```json
{
  "mongodbUri": "mongodb://mongo-user-mygeobot:password@195.80.51.100:27017/mygeobot",
  "telegramToken": "your-telegram-bot-token", 
  "webhookUrl": "https://tc-app.ru/webhook", 
  "port": 3030, 
  "domain": "tc-app.ru", 
  "email": "shkolnynick@gmail.com"
}
```

## 2. Создание CLI-приложения

Создадим файл `deploy.js`, который будет содержать код нашего CLI-приложения:

```javascript
// file: deploy.js
const fs = require('fs').promises; // Подключаем библиотеку fs для работы с файловой системой
const { exec } = require('child_process'); // Подключаем библиотеку exec для выполнения команд в терминале
const path = require('path'); // Подключаем библиотеку path для работы с путями файловой системы

// Путь к файлу конфигурации
const configPath = path.join(__dirname, 'config.json'); // Определяем путь к файлу конфигурации

// Функция для выполнения команд в терминале
const execPromise = (command) => {
    return new Promise((resolve, reject) => {
        exec(command, (error, stdout, stderr) => {
            if (error) {
                console.error(`Ошибка выполнения команды "${command}": ${error.message}`); // Обрабатываем ошибку выполнения
                return reject(error); // Возвращаем ошибку
            }
            console.log(`Команда выполнена: ${stdout}`); // Выводим сообщение об успешном выполнении
            resolve(stdout); // Возвращаем результат выполнения
        });
    });
};

// Функция для установки необходимых пакетов
const installPackages = async () => {
    await execPromise('sudo apt update && sudo apt install -y nginx nodejs npm certbot python3-certbot-nginx'); // Устанавливаем необходимые пакеты
};

// Функция для настройки Nginx
const setupNginx = async (config) => {
    const nginxConfig = `
server {
    listen 80; # Слушаем на порту 80
    listen [::]:80; # Слушаем на IPv6

    server_name ${config.domain}; # Указываем домен
    return 301 https://$server_name$request_uri; # Перенаправляем на HTTPS

    server_tokens off; # Отключаем отображение версии сервера
}

server {
    listen 443 ssl http2; # Слушаем на порту 443 с поддержкой SSL и HTTP/2
    listen [::]:443 ssl http2; # Слушаем на IPv6

    server_name ${config.domain}; # Указываем домен
    ssl_certificate /etc/letsencrypt/live/${config.domain}/fullchain.pem; # Путь к сертификату
    ssl_certificate_key /etc/letsencrypt/live/${config.domain}/privkey.pem; # Путь к ключу
    include /etc/letsencrypt/options-ssl-nginx.conf; # Включаем настройки SSL
    ssl_dhparam /etc/letsencrypt/ssl-dhparams.pem; # Включаем параметры DH

    location /webhook { # Настраиваем проксирование для вебхука
          proxy_pass http://127.0.0.1:${config.port}; # Проксируем на локальный порт
          proxy_set_header Host $host; # Устанавливаем заголовок Host
          proxy_set_header X-Real-IP $remote_addr; # Устанавливаем реальный IP
          proxy_set_header X-Forwarded-For $remote_addr; # Устанавливаем заголовок X-Forwarded-For
          proxy_set_header HTTPS $https; # Устанавливаем заголовок HTTPS
          proxy_set_header Scheme $scheme; # Устанавливаем заголовок Scheme
        }
}
`;

    await fs.writeFile(`/etc/nginx/sites-available/${config.domain}`, nginxConfig); // Записываем конфигурацию в файл
    await execPromise(`sudo ln -s /etc/nginx/sites-available/${config.domain} /etc/nginx/sites-enabled/`); // Создаем символическую ссылку
    await execPromise('sudo nginx -t'); // Проверяем конфигурацию Nginx
    await execPromise('sudo systemctl restart nginx'); // Перезапускаем Nginx
};

// Функция для получения сертификатов от Let's Encrypt
const obtainCertificates = async (config) => {
    await execPromise(`sudo certbot --nginx -d ${config.domain} --non-interactive --agree-tos --email ${config.email}`); // Получаем сертификаты
};

// Функция для установки зависимостей Node.js
const installNodeDependencies = async () => {
    await execPromise('npm install'); // Устанавливаем зависимости Node.js
};

// Функция для запуска приложения
const startApp = async () => {
    await execPromise(`node botApp.js & node notificationApp.js`); // Запускаем приложения
};

// Основная функция
const main = async () => {
    try {
        const config = JSON.parse(await fs.readFile(configPath, 'utf8')); // Читаем файл конфигурации
        await installPackages(); // Устанавливаем необходимые пакеты
        await setupNginx(config); // Настраиваем Nginx
        await obtainCertificates(config); // Получаем сертификаты от Let's Encrypt
        await installNodeDependencies(); // Устанавливаем зависимости Node.js
        await startApp(); // Запускаем приложение
        console.log('Деплой завершен успешно.'); // Выводим сообщение об успешном завершении
    } catch (error) {
        console.error('Ошибка в процессе деплоя:', error); // Обрабатываем ошибки
    }
};

main(); // Запускаем основную функцию
```

## 3. Инструкции по запуску



1. **Создаем файл конфигурации**: Создаем файл `config.json` в корне нашего проекта с необходимыми переменными окружения.

2. **Сохраняем код**: Сохраняем код CLI-приложения в файл `deploy.js`.


3. **Запускаем CLI-приложение**: В терминале выполняем команду:
   ```bash
   sudo node deploy.js
   ```
